# Scenario

**Feature**: compare the generated Grok export with the run of `[model.*]` tables in `config.toml`, read-only.

```
# config.toml + export -> locate the [model.*] run -> reduce the export to the same run -> report
config.toml, export -> ConfigWebGrokDiff -> state, region, preview, changes
```

## Preconditions

- Every leaf gets a fresh `t.TempDir()` holding `config.toml`; no leaf resolves the
  developer's real `~/.grok/config.toml`, changes directory, or sets an env var.
- The fixture mirrors a real `config.toml`: user sections (`[hints]`, `[cli]`)
  around the run, a `[models]` defaults table, and a `[model_providers.*]` table
  whose header merely starts like a model table.
- The export carries the prose header the CLI prints (`# Paste into …`), which is
  not part of the region.
- The Grok target is compare-only: `ConfigWebGrokDiff`
  (`open_ai/doctest_seam.go`) delegates to the same engine the report uses, with
  the path passed in, and writes nothing.

## Steps

1. A leaf sets `req.Content` (the file body; empty means absent) and
   `req.Preview` (the export).
2. `Run` writes the file when there is content, capturing its bytes, mode, and
   mtime.
3. `Run` compares through `ConfigWebGrokDiff`.
4. `Run` re-captures the bytes, mode, mtime, and the directory listing.
5. The leaf `Assert` checks the located run, the reduced preview, and that nothing
   was written.

## Context

- The run is the contiguous span of `[model.<id>]` tables: it starts at the first
  one and ends at the last line of the run.
- `[models]`, `[model_providers.*]`, and `[[…]]` headers belong to the user.
- Quoted and bare model keys both count.
- An absent file, or one without model tables, reports `missing` and is never
  created.
- `Changes` counts added plus removed lines between the two regions.

func Setup(t *testing.T, d *session.Doctest, req *Request) error {
	t.Helper()

// assertNoError fails if err is non-nil.
func assertNoError(t *testing.T, err error) {
	t.Helper()
	if err != nil {
		t.Fatalf("unexpected error: %v", err)
	}
}

// assertNoError is the only error helper this tree needs: the Grok target is
// compare-only, so no leaf expects a failure.

// readFileOrEmpty returns the file's text, or "" when it is absent.
func readFileOrEmpty(t *testing.T, path string) string {
	t.Helper()
	data, err := os.ReadFile(path)
	if err != nil {
		return ""
	}
	return string(data)
}

// statOf returns the file's permission string and mtime, or zero values when the
// file is absent.
func statOf(t *testing.T, path string) (string, string) {
	t.Helper()
	info, err := os.Stat(path)
	if err != nil {
		return "", ""
	}
	return fmt.Sprintf("%03o", info.Mode().Perm()), info.ModTime().UTC().Format("2006-01-02T15:04:05.000000000Z07:00")
}

// listDir returns the sorted base names in dir, so a leaf can assert that no
// backup or lock appeared.
func listDir(t *testing.T, dir string) []string {
	t.Helper()
	entries, err := os.ReadDir(dir)
	if err != nil {
		t.Fatalf("read dir: %v", err)
	}
	names := make([]string, 0, len(entries))
	for _, entry := range entries {
		names = append(names, entry.Name())
	}
	sort.Strings(names)
	return names
}

// tableBlocks returns the body lines of each [model.<id>] table in order.
func tableBlocks(text string) map[string]string {
	tables := map[string]string{}
	id := ""
	var body []string
	flush := func() {
		if id != "" {
			tables[id] = strings.Join(body, "\n")
		}
		id, body = "", nil
	}
	for _, line := range strings.Split(text, "\n") {
		if strings.HasPrefix(line, "[") {
			flush()
			if strings.HasPrefix(line, modelPrefix) && !strings.HasPrefix(line, "[[") {
				id = strings.TrimSuffix(strings.TrimPrefix(line, modelPrefix), "]")
				id = strings.Trim(id, `"`)
			}
			continue
		}
		if id != "" && strings.TrimSpace(line) != "" {
			body = append(body, strings.TrimSpace(line))
		}
	}
	flush()
	return tables
}

// tableIDs returns the sorted model ids in a document.
func tableIDs(text string) []string {
	blocks := tableBlocks(text)
	ids := make([]string, 0, len(blocks))
	for id := range blocks {
		ids = append(ids, id)
	}
	sort.Strings(ids)
	return ids
}
