import (
	"strings"
	"testing"

	"github.com/xhd2015/doctest/session"
)

func Assert(t *testing.T, d *session.Doctest, req *Request, resp *Response, err error) {
	t.Helper()
	assertNoError(t, err)
	for _, prose := range []string{"# Paste into", "# Generated by", "# Endpoint"} {
		if strings.Contains(resp.Diff.Preview, prose) {
			t.Fatalf("preview region kept %q:\n%s", prose, resp.Diff.Preview)
		}
	}
	if !strings.HasPrefix(resp.Diff.Preview, `[model."ais-glm-5-2"]`) {
		t.Fatalf("preview region must start at the first model table:\n%s", resp.Diff.Preview)
	}
	if got := tableIDs(resp.Diff.Preview); len(got) != 2 {
		t.Fatalf("preview region holds %v, want both generated tables", got)
	}
}
